     1→/**
     2→ * Production-ready Rule-Based Calendar (RBC) engine for TypeScript.
     3→ *
     4→ * Scope implemented from the provided RBC semantics:
     5→ * - Specific dates
     6→ * - Weekday rules
     7→ * - Monthday rules
     8→ * - Advanced combination of months + weekdays + monthdays
     9→ * - Base calendar-aware execution rules
    10→ * - Confirmation calendar + exception policy + shift-by
    11→ * - Activity period filtering
    12→ *
    13→ * Important assumptions:
    14→ * - Week starts on Monday.
    15→ * - D/-D/L/-L rules are evaluated against working days when a base calendar is supplied.
    16→ *   Without a base calendar, plain date semantics are used.
    17→ * - Exclusion rules always override inclusion rules.
    18→ * - Periodic calendar variants such as D3PA / L3PA are intentionally not implemented.
    19→ *   If you need periodic calendar support later, that should be added explicitly with a separate
    20→ *   period model instead of overloading regular-calendar logic.
    21→ */
    22→export class ScheduleValidationError extends Error {
    23→    details;
    24→    constructor(message, details = []) {
    25→        super(message);
    26→        this.name = 'ScheduleValidationError';
    27→        this.details = details;
    28→    }
    29→}
    30→const MONTH_NAMES = ['JAN', 'FEB', 'MAR', 'APR', 'MAY', 'JUN', 'JUL', 'AUG', 'SEP', 'OCT', 'NOV', 'DEC'];
    31→const VALID_MONTHS = new Set([...MONTH_NAMES, 'ALL']);
    32→const VALID_WEEK_MONTH_RELATIONS = new Set(['AND', 'OR']);
    33→const VALID_ACTIVITY_MODES = new Set(['ALWAYS', 'ACTIVE', 'NOT_ACTIVE']);
    34→const VALID_EXCEPTION_POLICIES = new Set([
    35→    'DISABLE_RUN',
    36→    'RUN_ON_NEXT_CONFIRMED_DAY_AND_SHIFT',
    37→    'RUN_ON_PREVIOUS_CONFIRMED_DAY_AND_SHIFT',
    38→    'RUN_IGNORE_CONFIRMATION_AND_SHIFT',
    39→    'SHIFT_AND_DISABLE_RUN',
    40→    'SHIFT_AND_RUN_ON_NEXT_CONFIRMED_DAY',
    41→    'SHIFT_AND_RUN_ON_PREVIOUS_CONFIRMED_DAY',
    42→    'SHIFT_AND_RUN_IGNORE_CONFIRMATION',
    43→]);
    44→export class ScheduleConfig {
    45→    CALENDARS;
    46→    SPECIFIC_DATES;
    47→    WEEKDAYS_CALENDAR;
    48→    WEEKDAYS;
    49→    WEEK_MONTH_RELATION;
    50→    MONTH_CALENDAR;
    51→    MONTHDAYS;
    52→    MONTHS;
    53→    CONFIRMATION;
    54→    ACTIVITY_PERIOD;
    55→    constructor(config = {}) {

    59→        this.WEEKDAYS = normalizeStringArray(config.WEEKDAYS || []);
    60→        this.WEEK_MONTH_RELATION = (config.WEEK_MONTH_RELATION || 'OR').toUpperCase();
    61→        this.MONTH_CALENDAR = config.MONTH_CALENDAR ?? null;
    62→        this.MONTHDAYS = normalizeStringArray(config.MONTHDAYS || []);
    63→        this.MONTHS = normalizeMonths(config.MONTHS || []);
    64→        this.CONFIRMATION = normalizeConfirmation(config.CONFIRMATION || null);
    65→        this.ACTIVITY_PERIOD = normalizeActivityPeriod(config.ACTIVITY_PERIOD || null);
    66→    }
    67→    update(newConfig = {}) {
    68→        if ('CALENDARS' in newConfig)
    69→            this.CALENDARS = normalizeCalendars(newConfig.CALENDARS || {});
    70→        if ('SPECIFIC_DATES' in newConfig)
    71→            this.SPECIFIC_DATES = normalizeSpecificDates(newConfig.SPECIFIC_DATES || []);
    72→        if ('WEEKDAYS_CALENDAR' in newConfig)
    73→            this.WEEKDAYS_CALENDAR = newConfig.WEEKDAYS_CALENDAR ?? null;
    74→        if ('WEEKDAYS' in newConfig)
    75→            this.WEEKDAYS = normalizeStringArray(newConfig.WEEKDAYS || []);
    76→        if ('WEEK_MONTH_RELATION' in newConfig)
    77→            this.WEEK_MONTH_RELATION = (newConfig.WEEK_MONTH_RELATION || 'OR').toUpperCase();
    78→        if ('MONTH_CALENDAR' in newConfig)
    79→            this.MONTH_CALENDAR = newConfig.MONTH_CALENDAR ?? null;
    80→        if ('MONTHDAYS' in newConfig)
    81→            this.MONTHDAYS = normalizeStringArray(newConfig.MONTHDAYS || []);
    82→        if ('MONTHS' in newConfig)
    83→            this.MONTHS = normalizeMonths(newConfig.MONTHS || []);
    84→        if ('CONFIRMATION' in newConfig)
    85→            this.CONFIRMATION = normalizeConfirmation(newConfig.CONFIRMATION || null);

    88→    }

    90→        this.CALENDARS[calendarName] = normalizeCalendar(calendarName, { WORKDAYS: workdays, HOLIDAYS: holidays });
    91→    }
    92→    validate() {
    93→        return validateConfig(this);
    94→    }
    95→    isDateEligible(date) {
    96→        const validation = validateConfig(this);
    97→        if (!validation.isValid) {
    98→            throw new ScheduleValidationError('Invalid schedule configuration', validation.errors);
    99→        }
   100→        const parsedDate = toUtcDate(date);
   101→        return evaluateScheduledDate(parsedDate, this) !== null;
   102→    }
   103→    getEligibleDates(year = new Date().getUTCFullYear()) {
   104→        return getEligibleRunDates(this, year);
   105→    }
   106→}
   107→export function validateConfig(config) {
   108→    const errors = [];
   109→    const warnings = [];
   110→    if (!config || typeof config !== 'object') {
   111→        return { isValid: false, errors: ['Configuration must be an object'], warnings: [] };
   112→    }
   113→    if (config.CALENDARS && (typeof config.CALENDARS !== 'object' || Array.isArray(config.CALENDARS))) {
   114→        errors.push('CALENDARS must be an object');
   115→    }
   116→    else if (config.CALENDARS) {
   117→        for (const [calendarName, calendar] of Object.entries(config.CALENDARS)) {
   118→            validateCalendar(calendarName, calendar, errors);
   119→        }
   120→    }
   121→    if (config.WEEKDAYS_CALENDAR && !config.CALENDARS?.[config.WEEKDAYS_CALENDAR]) {
   122→        warnings.push(`WEEKDAYS_CALENDAR "${config.WEEKDAYS_CALENDAR}" not found in CALENDARS; falling back to standard Mon-Fri workdays.`);
   123→    }
   124→    if (config.MONTH_CALENDAR && !config.CALENDARS?.[config.MONTH_CALENDAR]) {
   125→        warnings.push(`MONTH_CALENDAR "${config.MONTH_CALENDAR}" not found in CALENDARS; falling back to standard Mon-Fri workdays.`);
   126→    }
   127→    if (config.CONFIRMATION) {
   128→        const { CALENDAR, EXCEPTION_POLICY, SHIFT_BY } = config.CONFIRMATION;
   129→        if (!CALENDAR || !config.CALENDARS?.[CALENDAR]) {
   130→            warnings.push(`CONFIRMATION.CALENDAR "${CALENDAR}" not found in CALENDARS; falling back to standard Mon-Fri workdays.`);
   131→        }
   132→        if (EXCEPTION_POLICY && !VALID_EXCEPTION_POLICIES.has(EXCEPTION_POLICY)) {
   133→            errors.push(`Invalid CONFIRMATION.EXCEPTION_POLICY "${EXCEPTION_POLICY}"`);
   134→        }
   135→        if (SHIFT_BY === undefined || !Number.isInteger(SHIFT_BY) || SHIFT_BY < -62 || SHIFT_BY > 62) {
   136→            errors.push('CONFIRMATION.SHIFT_BY must be an integer between -62 and 62');
   137→        }
   138→    }
   139→    if (config.WEEK_MONTH_RELATION && !VALID_WEEK_MONTH_RELATIONS.has(config.WEEK_MONTH_RELATION)) {
   140→        errors.push('WEEK_MONTH_RELATION must be either "AND" or "OR"');
   141→    }
   142→    if (config.MONTHS && !Array.isArray(config.MONTHS)) {
   143→        errors.push('MONTHS must be an array');
   144→    }
   145→    else {
   146→        for (const month of config.MONTHS || []) {
   147→            if (!VALID_MONTHS.has(String(month).toUpperCase())) {
   148→                errors.push(`Invalid month "${month}"`);
   149→            }
   150→        }
   151→    }
   152→    if (config.SPECIFIC_DATES && !Array.isArray(config.SPECIFIC_DATES)) {
   153→        errors.push('SPECIFIC_DATES must be an array');
   154→    }
   155→    else {
   156→        for (const value of config.SPECIFIC_DATES || []) {
   157→            if (!isValidDateString(value)) {
   158→                errors.push(`Invalid specific date "${value}". Expected YYYY-MM-DD`);
   159→            }
   160→        }
   161→    }
   162→    validateRuleArray(config.WEEKDAYS, 'WEEKDAYS', parseWeekdayRule, errors);
   163→    validateRuleArray(config.MONTHDAYS, 'MONTHDAYS', parseMonthdayRule, errors);
   164→    if (config.ACTIVITY_PERIOD) {
   165→        const ap = config.ACTIVITY_PERIOD;
   166→        if (ap.MODE && !VALID_ACTIVITY_MODES.has(ap.MODE)) {
   167→            errors.push(`Invalid ACTIVITY_PERIOD.MODE "${ap.MODE}"`);
   168→        }
   169→        if (ap.MODE !== 'ALWAYS') {
   170→            if (!ap.FROM || !isValidDateString(ap.FROM))
   171→                errors.push('ACTIVITY_PERIOD.FROM must be a valid YYYY-MM-DD date');
   172→            if (!ap.TO || !isValidDateString(ap.TO))
   173→                errors.push('ACTIVITY_PERIOD.TO must be a valid YYYY-MM-DD date');
   174→            if (ap.FROM && ap.TO && isValidDateString(ap.FROM) && isValidDateString(ap.TO) && ap.FROM > ap.TO) {
   175→                errors.push('ACTIVITY_PERIOD.FROM must be less than or equal to ACTIVITY_PERIOD.TO');
   176→            }
   177→        }
   178→    }
   179→    if ((!config.SPECIFIC_DATES || config.SPECIFIC_DATES.length === 0) &&
   180→        (!config.WEEKDAYS || config.WEEKDAYS.length === 0) &&
   181→        (!config.MONTHDAYS || config.MONTHDAYS.length === 0)) {
   182→        warnings.push('No SPECIFIC_DATES, WEEKDAYS, or MONTHDAYS defined. All allowed days in selected months will match.');
   183→    }
   184→    if (!config.CALENDARS || Object.keys(config.CALENDARS).length === 0) {
   185→        warnings.push('No calendars defined. Calendar-based rules fall back to plain date semantics.');
   186→    }
   187→    return {
   188→        isValid: errors.length === 0,
   189→        errors,
   190→        warnings,
   191→    };
   192→}
   193→export function getEligibleRunDates(schedulingConfig, year = new Date().getUTCFullYear()) {
   194→    const validation = validateConfig(schedulingConfig);
   195→    if (!validation.isValid) {
   196→        return validation;
   197→    }
   198→    if (!Number.isInteger(year) || year < 1970 || year > 9999) {
   199→        return {
   200→            isValid: false,
   201→            errors: ['Year must be an integer between 1970 and 9999'],
   202→            warnings: [],
   203→        };
   204→    }
   205→    const eligibleSet = new Set();
   206→    const startDate = makeUtcDate(year, 0, 1);
   207→    const endDate = makeUtcDate(year, 11, 31);
   208→    for (let current = startDate; current <= endDate; current = addDays(current, 1)) {
   209→        const eligibleDate = evaluateScheduledDate(current, schedulingConfig);
   210→        if (eligibleDate) {
   211→            eligibleSet.add(formatDate(eligibleDate));
   212→        }
   213→    }
   214→    return Array.from(eligibleSet)
   215→        .sort()
   216→        .map((d) => parseDateString(d));
   217→}
   218→export function evaluateScheduledDate(date, config) {
   219→    if (!matchesMonth(date, config.MONTHS))
   220→        return null;
   221→    if (!isActivityAllowed(date, config.ACTIVITY_PERIOD || null))
   222→        return null;
   223→    const specificDateMatch = (config.SPECIFIC_DATES || []).includes(formatDate(date));
   224→    const weekdayMatch = evaluateWeekdayDimension(date, config);
   225→    const monthdayMatch = evaluateMonthdayDimension(date, config);
   226→    let baseMatch;
   227→    if (specificDateMatch) {
   228→        baseMatch = true;
   229→    }
   230→    else if ((!config.WEEKDAYS || config.WEEKDAYS.length === 0) && (!config.MONTHDAYS || config.MONTHDAYS.length === 0)) {
   231→        baseMatch = Boolean(config.SPECIFIC_DATES && config.SPECIFIC_DATES.length > 0);
   232→    }
   233→    else if (config.WEEKDAYS && config.WEEKDAYS.length > 0 && (!config.MONTHDAYS || config.MONTHDAYS.length === 0)) {
   234→        baseMatch = weekdayMatch;
   235→    }
   236→    else if ((!config.WEEKDAYS || config.WEEKDAYS.length === 0) && config.MONTHDAYS && config.MONTHDAYS.length > 0) {
   237→        baseMatch = monthdayMatch;
   238→    }
   239→    else {
   240→        baseMatch =
   241→            config.WEEK_MONTH_RELATION === 'AND' ? weekdayMatch && monthdayMatch : weekdayMatch || monthdayMatch;
   242→    }
   243→    if (!baseMatch)
   244→        return null;
   245→    return applyConfirmationPolicy(date, config);
   246→}
   247→function evaluateWeekdayDimension(date, config) {
   248→    if (!config.WEEKDAYS || config.WEEKDAYS.length === 0)
   249→        return false;
   250→    const calendar = config.WEEKDAYS_CALENDAR && config.CALENDARS?.[config.WEEKDAYS_CALENDAR]
   251→        ? normalizeCalendar(config.WEEKDAYS_CALENDAR, config.CALENDARS[config.WEEKDAYS_CALENDAR])
   252→        : null;
   253→    const rules = config.WEEKDAYS.map(parseWeekdayRule);
   254→    for (const rule of rules) {
   255→        if (rule.modifier === '-' && matchesWeekdayRule(date, rule, calendar)) {
   256→            return false;
   257→        }
   258→    }
   259→    for (const rule of rules) {
   260→        if (rule.modifier !== '-' && matchesWeekdayRule(date, rule, calendar)) {
   261→            return true;
   262→        }
   263→    }
   264→    return false;
   265→}
   266→function evaluateMonthdayDimension(date, config) {
   267→    if (!config.MONTHDAYS || config.MONTHDAYS.length === 0)
   268→        return false;
   269→    const calendar = config.MONTH_CALENDAR && config.CALENDARS?.[config.MONTH_CALENDAR]
   270→        ? normalizeCalendar(config.MONTH_CALENDAR, config.CALENDARS[config.MONTH_CALENDAR])
   271→        : null;
   272→    const rules = config.MONTHDAYS.map(parseMonthdayRule);
   273→    for (const rule of rules) {
   274→        if (rule.modifier === '-' && matchesMonthdayRule(date, rule, calendar)) {
   275→            return false;
   276→        }
   277→    }
   278→    for (const rule of rules) {
   279→        if (rule.modifier !== '-' && matchesMonthdayRule(date, rule, calendar)) {
   280→            return true;
   281→        }
   282→    }
   283→    return false;
   284→}
   285→function matchesWeekdayRule(candidateDate, rule, calendar) {
   286→    switch (rule.type) {
   287→        case 'DAY_OF_WEEK': {
   288→            const selectedDay = getDayOfWeekInSameWeek(candidateDate, rule.dayOfWeek);
   289→            return applyWorkingDayModifierMatch(candidateDate, selectedDay, rule.modifier, calendar);
   290→        }
   291→        case 'NTH_WORKING_DAY_OF_WEEK': {
   292→            if (!calendar) {
   293→                return getDayIndexFromStartOfWeek(candidateDate) === rule.n;
   294→            }
   295→            const nthDate = getNthWorkingDayOfWeek(candidateDate, rule.n, calendar);
   296→            return nthDate !== null && formatDate(candidateDate) === formatDate(nthDate);
   297→        }
   298→        case 'DAY_OF_WEEK_IN_WEEK_OF_MONTH': {
   299→            if (getWeekOfMonth(candidateDate) !== rule.weekOfMonth)
   300→                return false;
   301→            const selectedDay = getDayOfWeekInSameWeek(candidateDate, rule.dayOfWeek);
   302→            return applyWorkingDayModifierMatch(candidateDate, selectedDay, rule.modifier, calendar);
   303→        }
   304→        case 'NTH_FROM_END_OF_WORKING_WEEK': {
   305→            if (!calendar) {
   306→                return getDayIndexFromEndOfWeek(candidateDate) === rule.n;
   307→            }
   308→            const target = getNthWorkingDayFromEndOfWeek(candidateDate, rule.n, calendar);
   309→            return target !== null && formatDate(candidateDate) === formatDate(target);
   310→        }
   311→        default:
   312→            return false;
   313→    }
   314→}
   315→function matchesMonthdayRule(candidateDate, rule, calendar) {
   316→    switch (rule.type) {
   317→        case 'ALL_WORKING_DAYS': {
   318→            return !calendar || isWorkingDay(candidateDate, calendar);
   319→        }
   320→        case 'DAY_OF_MONTH': {
   321→            const daysInMonth = getDaysInMonth(candidateDate);
   322→            if (rule.dayOfMonth > daysInMonth)
   323→                return false;
   324→            const selectedDay = makeUtcDate(candidateDate.getUTCFullYear(), candidateDate.getUTCMonth(), rule.dayOfMonth);
   325→            return applyWorkingDayModifierMatch(candidateDate, selectedDay, rule.modifier, calendar);
   326→        }
   327→        case 'NTH_WORKING_DAY_OF_MONTH': {
   328→            if (!calendar) {
   329→                return candidateDate.getUTCDate() === rule.n;
   330→            }
   331→            const nthDate = getNthWorkingDayOfMonth(candidateDate, rule.n, calendar);
   332→            return nthDate !== null && formatDate(candidateDate) === formatDate(nthDate);
   333→        }
   334→        case 'NTH_FROM_END_OF_WORKING_MONTH': {
   335→            if (!calendar) {
   336→                const daysInMonth = getDaysInMonth(candidateDate);
   337→                return candidateDate.getUTCDate() === daysInMonth - rule.n + 1;
   338→            }
   339→            const target = getNthWorkingDayFromEndOfMonth(candidateDate, rule.n, calendar);
   340→            return target !== null && formatDate(candidateDate) === formatDate(target);
   341→        }
   342→        default:
   343→            return false;
   344→    }
   345→}
   346→function applyWorkingDayModifierMatch(candidateDate, selectedDay, modifier, calendar) {
   347→    if (!selectedDay)
   348→        return false;
   349→    if (!calendar) {
   350→        return formatDate(candidateDate) === formatDate(selectedDay);
   351→    }
   352→    if (modifier === '>') {
   353→        const effective = isWorkingDay(selectedDay, calendar) ? selectedDay : getNextWorkingDay(selectedDay, calendar);
   354→        return effective !== null && formatDate(candidateDate) === formatDate(effective);
   355→    }
   356→    if (modifier === '<') {
   357→        const effective = isWorkingDay(selectedDay, calendar) ? selectedDay : getPreviousWorkingDay(selectedDay, calendar);
   358→        return effective !== null && formatDate(candidateDate) === formatDate(effective);
   359→    }
   360→    return formatDate(candidateDate) === formatDate(selectedDay) && isWorkingDay(selectedDay, calendar);
   361→}
   362→function applyConfirmationPolicy(date, config) {
   363→    if (!config.CONFIRMATION)
   364→        return date;
   365→    const { CALENDAR, EXCEPTION_POLICY, SHIFT_BY } = config.CONFIRMATION;
   366→    if (!CALENDAR || !config.CALENDARS?.[CALENDAR])
   367→        return date;
   368→    const calendar = normalizeCalendar(CALENDAR, config.CALENDARS[CALENDAR]);
   369→    const confirmed = isWorkingDay(date, calendar);
   370→    const shiftAmount = SHIFT_BY ?? 0;
   371→    switch (EXCEPTION_POLICY) {
   372→        case 'DISABLE_RUN':
   373→            return confirmed ? date : null;
   374→        case 'RUN_ON_NEXT_CONFIRMED_DAY_AND_SHIFT': {
   375→            const base = confirmed ? date : getNextWorkingDay(date, calendar);
   376→            return base ? addDays(base, shiftAmount) : null;
   377→        }
   378→        case 'RUN_ON_PREVIOUS_CONFIRMED_DAY_AND_SHIFT': {
   379→            const base = confirmed ? date : getPreviousWorkingDay(date, calendar);
   380→            return base ? addDays(base, shiftAmount) : null;
   381→        }
   382→        case 'RUN_IGNORE_CONFIRMATION_AND_SHIFT':
   383→            return addDays(date, shiftAmount);
   384→        case 'SHIFT_AND_DISABLE_RUN': {
   385→            const shifted = addDays(date, shiftAmount);
   386→            return isWorkingDay(shifted, calendar) ? shifted : null;
   387→        }
   388→        case 'SHIFT_AND_RUN_ON_NEXT_CONFIRMED_DAY': {
   389→            const shifted = addDays(date, shiftAmount);
   390→            return isWorkingDay(shifted, calendar) ? shifted : getNextWorkingDay(shifted, calendar);
   391→        }
   392→        case 'SHIFT_AND_RUN_ON_PREVIOUS_CONFIRMED_DAY': {
   393→            const shifted = addDays(date, shiftAmount);
   394→            return isWorkingDay(shifted, calendar) ? shifted : getPreviousWorkingDay(shifted, calendar);
   395→        }
   396→        case 'SHIFT_AND_RUN_IGNORE_CONFIRMATION':
   397→            return addDays(date, shiftAmount);
   398→        default:
   399→            return date;
   400→    }
   401→}
   402→export function parseWeekdayRule(rawRule) {
   403→    const rule = normalizeRule(rawRule);
   404→    const modifier = extractModifier(rule);
   405→    const body = stripModifier(rule);
   406→    if (/^[1-7]$/.test(body)) {
   407→        return { modifier, type: 'DAY_OF_WEEK', dayOfWeek: Number(body) };
   408→    }
   409→    if (/^D[1-7]$/.test(body)) {
   410→        return { modifier, type: 'NTH_WORKING_DAY_OF_WEEK', n: Number(body.slice(1)) };
   411→    }
   412→    if (/^D[1-7]W[1-6]$/.test(body)) {
   413→        return {
   414→            modifier,
   415→            type: 'DAY_OF_WEEK_IN_WEEK_OF_MONTH',
   416→            dayOfWeek: Number(body[1]),
   417→            weekOfMonth: Number(body[3]),
   418→        };
   419→    }
   420→    if (/^L[1-7]$/.test(body)) {
   421→        return { modifier, type: 'NTH_FROM_END_OF_WORKING_WEEK', n: Number(body.slice(1)) };
   422→    }
   423→    throw new Error(`Unsupported weekday rule "${rawRule}"`);
   424→}
   425→export function parseMonthdayRule(rawRule) {
   426→    const rule = normalizeRule(rawRule);
   427→    const modifier = extractModifier(rule);
   428→    const body = stripModifier(rule);
   429→    if (body === 'WORKDAYS') {
   430→        return { modifier, type: 'ALL_WORKING_DAYS' };
   431→    }
   432→    if (/^([1-9]|[12][0-9]|3[01])$/.test(body)) {
   433→        return { modifier, type: 'DAY_OF_MONTH', dayOfMonth: Number(body) };
   434→    }
   435→    if (/^D([1-9]|[12][0-9]|3[01])$/.test(body)) {
   436→        return { modifier, type: 'NTH_WORKING_DAY_OF_MONTH', n: Number(body.slice(1)) };
   437→    }
   438→    if (/^L([1-9]|[12][0-9]|3[01])$/.test(body)) {
   439→        return { modifier, type: 'NTH_FROM_END_OF_WORKING_MONTH', n: Number(body.slice(1)) };
   440→    }
   441→    throw new Error(`Unsupported monthday rule "${rawRule}"`);
   442→}
   443→function validateRuleArray(rules, label, parser, errors) {
   444→    if (!rules)
   445→        return;
   446→    if (!Array.isArray(rules)) {
   447→        errors.push(`${label} must be an array`);
   448→        return;
   449→    }
   450→    for (const rule of rules) {
   451→        try {
   452→            parser(rule);
   453→        }
   454→        catch (err) {
   455→            errors.push(err instanceof Error ? err.message : String(err));
   456→        }
   457→    }
   458→}
   459→function validateCalendar(calendarName, calendar, errors) {
   460→    if (!calendar || typeof calendar !== 'object' || Array.isArray(calendar)) {
   461→        errors.push(`Calendar "${calendarName}" must be an object`);
   462→        return;
   463→    }
   464→    if (!Array.isArray(calendar.WORKDAYS)) {
   465→        errors.push(`Calendar "${calendarName}" WORKDAYS must be an array`);
   466→    }
   467→    else {
   468→        for (const day of calendar.WORKDAYS) {
   469→            if (!Number.isInteger(day) || day < 1 || day > 7) {
   470→                errors.push(`Calendar "${calendarName}" has invalid WORKDAY value "${day}". Valid values are 1-7 where 1=Mon and 7=Sun`);
   471→            }
   472→        }
   473→    }
   474→    if (!Array.isArray(calendar.HOLIDAYS)) {
   475→        errors.push(`Calendar "${calendarName}" HOLIDAYS must be an array`);
   476→    }
   477→    else {
   478→        for (const holiday of calendar.HOLIDAYS) {
   479→            if (!isValidDateString(holiday)) {
   480→                errors.push(`Calendar "${calendarName}" has invalid HOLIDAY "${holiday}". Expected YYYY-MM-DD`);
   481→            }
   482→        }
   483→    }
   484→}

   547→export function isActivityAllowed(date, activityPeriod) {
   548→    if (!activityPeriod || activityPeriod.MODE === 'ALWAYS')
   549→        return true;
   550→    if (!activityPeriod.FROM || !activityPeriod.TO)
   551→        return true;
   552→    const from = parseDateString(activityPeriod.FROM);
   553→    const to = parseDateString(activityPeriod.TO);
   554→    if (!from || !to)
   555→        return true;
   556→    const dateStr = formatDate(date);
   557→    const fromStr = formatDate(from);
   558→    const toStr = formatDate(to);
   559→    if (activityPeriod.MODE === 'ACTIVE') {
   560→        return dateStr >= fromStr && dateStr <= toStr;
   561→    }
   562→    if (activityPeriod.MODE === 'NOT_ACTIVE') {
   563→        return dateStr < fromStr || dateStr > toStr;
   564→    }
   565→    return true;
   566→}

   574→export function getNthWorkingDayOfWeek(date, n, calendar) {

   680→export function isValidDateString(value) {
   681→    if (typeof value !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(value))
   682→        return false;
   683→    const parsed = parseDateString(value);
   684→    return parsed !== null && formatDate(parsed) === value;
   685→}
   686→export function parseDateString(value) {
   687→    if (typeof value !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(value))
   688→        return null;
   689→    const [year, month, day] = value.split('-').map(Number);
   690→    const date = makeUtcDate(year, month - 1, day);
   691→    if (date.getUTCFullYear() !== year || date.getUTCMonth() !== month - 1 || date.getUTCDate() !== day) {
   692→        return null;
   693→    }
   694→    return date;
   695→}
   696→export function toUtcDate(value) {
   697→    if (value instanceof Date && !Number.isNaN(value.getTime())) {
   698→        return makeUtcDate(value.getUTCFullYear(), value.getUTCMonth(), value.getUTCDate());
   699→    }
   700→    if (typeof value === 'string') {
   701→        const parsed = parseDateString(value);
   702→        if (!parsed)
   703→            throw new Error(`Invalid date string "${value}". Expected YYYY-MM-DD`);
   704→        return parsed;
   705→    }
   706→    throw new Error('Date must be a valid Date object or YYYY-MM-DD string');
   707→}
   708→export function makeUtcDate(year, monthIndex, day) {
   709→    return new Date(Date.UTC(year, monthIndex, day));

   711→export function addDays(date, days) {

   715→}
